class Solution {
public:
    int calPoints(vector<string>& operations) {
        stack<int> myStack;
        int num1=0,num2=0;
        for (int i=0;i<operations.size();i++){
            if (operations[i][0]=='C')
                myStack.pop(); 
            else if (operations[i][0]=='D'){
                myStack.push(2*myStack.top());
            }
            else if (operations[i][0]=='+'){
                num1=myStack.top();
                myStack.pop();
                num2=myStack.top();
                myStack.push(num1);
                myStack.push(num1+num2);
            }
            else
                myStack.push(stoi(operations[i]));
        }
        num1=0;
        while (!myStack.empty()) {
            num1+=myStack.top();
            myStack.pop();
        }
        return num1;
    }
};



class Solution(object):
    def calPoints(self, operations):
        array=[]
        for i in range(len(operations)):
            if operations[i]=='C':
                array.pop()
            elif operations[i]=='D':
                num=array.pop()
                array.append(num)
                array.append(2*num)
            elif operations[i]=='+':
                num1=array.pop()
                num2=array.pop()
                array.append(num2)
                array.append(num1)
                array.append(num1+num2)
            else:
                array.append(int(operations[i]))
        num=0
        for i in array:
            num=num+i
        return num
