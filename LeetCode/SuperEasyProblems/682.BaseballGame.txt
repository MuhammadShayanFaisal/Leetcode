class Solution {
public:
    int calPoints(vector<string>& operations) {
        stack<int> myStack;
        int num1=0,num2=0;
        for (int i=0;i<operations.size();i++){
            if (operations[i][0]=='C')
                myStack.pop(); 
            else if (operations[i][0]=='D'){
                myStack.push(2*myStack.top());
            }
            else if (operations[i][0]=='+'){
                num1=myStack.top();
                myStack.pop();
                num2=myStack.top();
                myStack.push(num1);
                myStack.push(num1+num2);
            }
            else
                myStack.push(stoi(operations[i]));
        }
        num1=0;
        while (!myStack.empty()) {
            num1+=myStack.top();
            myStack.pop();
        }
        return num1;
    }
};
