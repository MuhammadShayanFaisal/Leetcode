class Solution {
public:
    int xorOperation(int n, int start) {
        int nums[n];
        int result=0;
        for (int i=0;i<n;i++){
            nums[i]=start+2*i;
            result^=nums[i];
        }
        return result;
    }
};


class Solution(object):
    def xorOperation(self, n, start):
        result=0
        nums=[]
        for i in range(n):
            nums.append(start+2*i)
            result=result^nums[i]
        return result
