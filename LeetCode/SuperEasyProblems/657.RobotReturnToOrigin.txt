class Solution {
public:
    bool judgeCircle(string moves) {
        int x=0,y=0;
        for (int i=0;i<moves.size();i++){
            if (moves[i]=='U')
            y+=1;
            else if (moves[i]=='D')
            y-=1;
            else if (moves[i]=='L')
            x+=1;
            else 
            x-=1;
        }
        if (x==0 && y==0)
        return true;
        return false;

    }
};




class Solution(object):
    def judgeCircle(self, moves):
        x=0
        y=0
        for i in moves:
            if (i=='U'):
                y+=1
            elif (i=='D'):
                y-=1
            elif (i=='L'):
                x+=1
            else: 
                x-=1
        if (x==0 and y==0):
            return True
        return False
