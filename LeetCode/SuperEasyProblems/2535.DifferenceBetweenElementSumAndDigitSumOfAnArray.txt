class Solution(object):
    def differenceOfSum(self, nums):
        e_Sum=0
        d_Sum=0
        for i in nums:
            e_Sum=e_Sum+i
            if i<9 :
                d_Sum=d_Sum+i
            else:
                num=len(str(abs(i)))
                for j in range(num):
                    d_Sum=d_Sum+i%10
                    i=i/10
        return abs(e_Sum-d_Sum)


class Solution {
public:
    int differenceOfSum(vector<int>& nums) {
        int e_Sum=0,d_Sum=0;
        for (int i=0;i<nums.size();i++){
            e_Sum+=nums[i];
            if (nums[i]<10)
            d_Sum+=nums[i];
            else{
                int len=log10(nums[i])+1;
                for (int j=0;j<len;j++){
                    d_Sum=d_Sum+nums[i]%10;
                    nums[i]/=10;
                }
            }
        }
        return abs(d_Sum-e_Sum);
    }
};
