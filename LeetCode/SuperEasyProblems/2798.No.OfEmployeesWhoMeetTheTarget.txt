class Solution {
public:
    int numberOfEmployeesWhoMetTarget(vector<int>& hours, int target) {
        int e=0;
        for (int i=0;i<hours.size();i++){
            if (hours[i]>=target)
            e++;
        }
        return e;
    }
};

class Solution(object):
    def numberOfEmployeesWhoMetTarget(self, hours, target):
        e=0
        for i in hours:
            if i>=target:
                e=e+1
        return e
