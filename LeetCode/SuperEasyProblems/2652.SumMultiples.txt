class Solution {
public:
    int sumOfMultiples(int n) {
        int sum=0;
        for (int i=1;i<=n;i++){
            if (i%3==0 || i%5==0 || i%7==0)
            sum+=i;
        }
        return sum;
    }
};

class Solution(object):
    def sumOfMultiples(self, n):
        sum=0
        for i in range(n):
            if (i+1)%3==0 or (i+1)%5==0 or (i+1)%7==0:
                sum=sum+i+1
        return sum
