class Solution {
public:
    int differenceOfSums(int n, int m) {
        int num1=0,num2=0;
        for (int i=1;i<=n;i++){
           if (i%m!=0)
           num1+=i;
           else
           num2+=i;
        }
        return num1-num2;
        
    }
};



class Solution(object):
    def differenceOfSums(self, n, m):
        num1=0
        num2=0
        for i in range(n):
           if (i+1)%m!=0:
            num1+=(i+1)
           else:

        return num1-num2
