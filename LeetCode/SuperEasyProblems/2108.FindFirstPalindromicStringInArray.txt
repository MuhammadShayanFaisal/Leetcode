class Solution(object):
    def firstPalindrome(self, words):
        """
        :type words: List[str]
        :rtype: str
        """
        for i in words:
            if i == i[::-1]:
                return i
        return ""

class Solution {
public:
    string firstPalindrome(vector<string>& words) {
        for (int i=0;i<words.size();i++){
            if (words[i][0]!=words[i][words[i].size()-1])
            continue;
            // else if(words[i][1]!=words[i][words[i].size()-2])
            // continue;
            else{
                if (words[i].size()<3)
                return words[i];
                // cout<<words[i]<<endl;
                int k=words[i].size()-2;
                for (int j=1;j<(words[i].size()/2)+1;j++){
                    // cout<<words[i][j]<<"?????"<<words[i][k]<<endl;
                    if (words[i][j]!=words[i][k])
                    break;
                    else{
                        if (j==k)
                        return words[i];
                        else if ((words[i].size()%2==0) && j+1==k)
                        return words[i];
                    }
                    k--;
                }
            }
        }
        return "";
    }
};
