class Solution {
public:
    int maximumWealth(vector<vector<int>>& accounts) {
        int amount[accounts.size()];
        int i=0;
        for (i=0;i<accounts.size();i++){
            amount[i]=0;
            for (int j=0;j<accounts[i].size();j++){
                amount[i]+=accounts[i][j];
            }
            if (i>0 && amount[i-1]>amount[i])
            amount[i]=amount[i-1];
        }
        return amount[i-1];
    }
};

class Solution(object):
    def maximumWealth(self, accounts):
        amount=[]
        for i in range(len(accounts)):
            amount.append(0)
            for j in range(len(accounts[i])):
                amount[i]=amount[i]+accounts[i][j]
            if (i>0 and amount[i-1]>amount[i]):
                amount[i]=amount[i-1]
        return amount[i-1]
