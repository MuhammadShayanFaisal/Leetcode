class Solution {
public:
    bool threeConsecutiveOdds(vector<int>& arr) {
        if (arr.size()<3)
        return false;
        for (int i=0;i<arr.size();i++){
            if(arr[i]%2!=0 && i==arr.size()-2)
            return false;
            else if(arr[i]%2!=0 && i<arr.size()-2){
                if (arr[i+1]%2!=0 && arr[i+2]%2!=0)
                return true;
            }
        }
        return false;
    }
};



class Solution(object):
    def threeConsecutiveOdds(self, arr):
        if len(arr)<3:
            return False
        for i in range(len(arr)):
            if arr[i]%2!=0 and i==len(arr)-2:
                return False 
            elif arr[i]%2!=0 and i<len(arr)-2:
                if arr[i+1]%2!=0 and arr[i+2]%2!=0:
                    return True
        return False
